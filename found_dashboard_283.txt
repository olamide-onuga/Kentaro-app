from flask import Blueprint, render_template, redirect, url_for, flash, request, jsonify
from flask_login import login_required, current_user
from datetime import date
from app.extensions import db
from app.models.user import ClientProfile
from app.models.coaching import CoachingPackage
from app.models.workout import Workout, WorkoutExercise
from app.models.progress import ClientProgram, WorkoutCompletion

client_bp = Blueprint('client', __name__, url_prefix='/client')

@client_bp.before_request
@login_required
def require_client():
    if not current_user.is_client:
        flash('You do not have permission to access this page.', 'danger')
        return redirect(url_for('main.index'))

@client_bp.route('/dashboard')
def dashboard():
    profile = current_user.client_profile
    today_workout = None
    active_program = None

    if profile:
        cp = profile.client_programs.filter_by(is_active=True).first()
        if cp:
            active_program = cp.program
            # Find today's workout — simple heuristic: find first uncompleted workout in order
            completed_ids = [c.workout_id for c in profile.completions.all()]
            for phase in cp.program.phases.order_by('order').all():
                for w in phase.workouts.order_by('week_number', 'day_number').all():
                    if w.id not in completed_ids:
                        today_workout = w
                        break
                if today_workout:
                    break

    return render_template('client/dashboard.html',
                           profile=profile,
                           today_workout=today_workout,
                           active_program=active_program)

@client_bp.route('/my-program')
def my_program():
    profile = current_user.client_profile
    cp = profile.client_programs.filter_by(is_active=True).first() if profile else None
    completed_ids = set(c.workout_id for c in profile.completions.all()) if profile else set()
    return render_template('client/my_program.html', cp=cp, completed_ids=completed_ids)

@client_bp.route('/workout/<int:workout_id>')
def workout_view(workout_id):
    profile = current_user.client_profile
    workout = Workout.query.get_or_404(workout_id)
    already_completed = WorkoutCompletion.query.filter_by(
        client_id=profile.id, workout_id=workout_id
    ).first() if profile else None
    return render_template('client/workout_view.html',
                           workout=workout,
                           already_completed=already_completed)

@client_bp.route('/workout/<int:workout_id>/complete', methods=['POST'])
def complete_workout(workout_id):
    profile = current_user.client_profile
    if not profile:
        flash('Profile not found.', 'danger')
        return redirect(url_for('client.dashboard'))

    # Prevent duplicate completions
    existing = WorkoutCompletion.query.filter_by(client_id=profile.id, workout_id=workout_id).first()
    if not existing:
        completion = WorkoutCompletion(
            client_id=profile.id,
            workout_id=workout_id,
            client_notes=request.form.get('notes'),
            perceived_effort=request.form.get('perceived_effort') or None
        )
        db.session.add(completion)
        db.session.commit()
        flash('Workout marked as complete! Great work! 💪', 'success')
    else:
        flash('You already completed this workout.', 'info')

    return redirect(url_for('client.dashboard'))

@client_bp.route('/history')
def history():
    profile = current_user.client_profile
    completions = profile.completions.order_by(WorkoutCompletion.completed_at.desc()).all() if profile else []
    return render_template('client/history.html', completions=completions)

@client_bp.route('/select-package/<int:package_id>', methods=['GET', 'POST'])
def select_package(package_id):
    package = CoachingPackage.query.get_or_404(package_id)

    if request.method == 'POST':
        profile = current_user.client_profile
        if not profile:
            profile = ClientProfile(user_id=current_user.id)
            db.session.add(profile)

        profile.coach_id = package.coach_id
        profile.active_package_id = package.id
        db.session.commit()

        flash(f'Successfully enrolled in {package.name}!', 'success')
        return redirect(url_for('client.dashboard'))

    return render_template('client/select_package.html', package=package)
